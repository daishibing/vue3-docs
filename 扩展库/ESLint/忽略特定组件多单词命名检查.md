# 忽略特定组件多单词命名检查

通过覆盖 ESLint 配置，忽略特定组件触发的 `vue/multi-word-component-names` 检查

# 配置 `eslint.config.ts`

```ts
import { defineConfigWithVueTs } from "@vue/eslint-config-typescript"

export default defineConfigWithVueTs(
    // 其他配置
    {
        files: ["src/pages/**/index.vue", "src/pages/**/[...path].vue", "src/pages/**/_parent.vue"],
        ignores: ["src/pages/**/components/**"],
        rules: {
            "vue/multi-word-component-names": "off",
        },
    },
)
```
